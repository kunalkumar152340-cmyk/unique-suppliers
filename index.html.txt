function renderProducts(){
  const grid = document.getElementById('productGrid');
  const country = document.getElementById('countryFilter').value;
  const q = document.getElementById('searchInput').value.trim().toLowerCase();
  let list = products.filter(p=>{
    const catOk = activeCat==='all' || p.cat===activeCat;
    const countryOk = country==='all' || p.country===country;
    const qOk = !q || (p.name+p.brand+p.model+p.category).toLowerCase().includes(q);
    return catOk && countryOk && qOk;
  });
  if(list.length===0){ grid.innerHTML = '<p style="color:var(--text-dim);">No machines match your filters.</p>'; return; }
  grid.innerHTML = list.map(p=>{
    const cat = categories.find(c=>c.id===p.cat);
    return `<div class="pcard">
      <div class="thumb">${cat.icon}<span class="country">${p.country}</span></div>
      <div class="body">
        <div class="cat-label">${cat.name}</div>
        <h4>${p.name}</h4>
        <div class="meta">
          <span>Brand: <b>${p.brand}</b></span>
          <span>Model: <b>${p.model}</b></span>
          <span>Capacity: <b>${p.capacity}</b></span>
        </div>
        <div class="row-bottom">
          <div class="price">Rs. ${p.price.toLocaleString()}<small>incl. 13% VAT</small></div>
          <button class="addbtn" data-add="${p.id}">+</button>
        </div>
      </div>
    </div>`;
  }).join('');
  grid.querySelectorAll('[data-add]').forEach(btn=>{
    btn.onclick = ()=> addToCart(parseInt(btn.dataset.add));
  });
}

function saveCart(){ sessionStorage.setItem('us_cart', JSON.stringify(cart)); }
function addToCart(id){
  const p = products.find(x=>x.id===id);
  const line = cart.find(c=>c.id===id);
  if(line) line.qty++; else cart.push({id:p.id, name:p.name, price:p.price, qty:1, icon: categories.find(c=>c.id===p.cat).icon});
  saveCart(); renderCart(); openDrawer();
}
function renderCart(){
  const body = document.getElementById('cartBody');
  document.getElementById('cartCount').textContent = cart.reduce((a,c)=>a+c.qty,0);
  if(cart.length===0){ body.innerHTML = '<p style="color:var(--text-dim); font-size:13px;">Your cart is empty.</p>'; }
  else{
    body.innerHTML = cart.map(c=>`
      <div class="cart-line">
        <div class="ic">${c.icon}</div>
        <div class="info"><b>${c.name}</b><span>Rs. ${c.price.toLocaleString()} each</span></div>
        <div class="qty-ctrl">
          <button data-dec="${c.id}">−</button><span>${c.qty}</span><button data-inc="${c.id}">+</button>
        </div>
      </div>`).join('');
    body.querySelectorAll('[data-inc]').forEach(b=>b.onclick=()=>changeQty(parseInt(b.dataset.inc),1));
    body.querySelectorAll('[data-dec]').forEach(b=>b.onclick=()=>changeQty(parseInt(b.dataset.dec),-1));
  }
  const total = cart.reduce((a,c)=>a+c.price*c.qty,0);
  const taxable = total/(1+VAT_RATE);
  const vat = total - taxable;
  document.getElementById('cartSubtotal').textContent = 'Rs. ' + total.toLocaleString(undefined,{maximumFractionDigits:0});
  document.getElementById('cartVat').textContent = 'Rs. ' + vat.toLocaleString(undefined,{maximumFractionDigits:0});
  document.getElementById('cartTotal').textContent = 'Rs. ' + total.toLocaleString(undefined,{maximumFractionDigits:0});
}
function changeQty(id, delta){
  const line = cart.find(c=>c.id===id);
  line.qty += delta;
  if(line.qty<=0) cart = cart.filter(c=>c.id!==id);
  saveCart(); renderCart();
}

let invoiceCounter = 1042;
function buildInvoice(name, phone, addr, method){
  const items = JSON.parse(sessionStorage.getItem('us_cart')||'[]');
  const total = items.reduce((a,c)=>a+c.price*c.qty,0);
  const taxable = total/(1+VAT_RATE);
  const vat = total - taxable;
  invoiceCounter++;
  const now = new Date();
  return {
    no: 'US-' + invoiceCounter,
    date: now.toLocaleDateString(),
    time: now.toLocaleTimeString(),
    name, phone, addr, method, items, total, taxable, vat
  };
}

function downloadCSV(content, filename){
  const blob = new Blob([content], {type:'text/csv'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a'); a.href = url; a.download = filename; a.click();
  URL.revokeObjectURL(url);
}
